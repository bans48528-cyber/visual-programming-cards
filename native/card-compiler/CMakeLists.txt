cmake_minimum_required(VERSION 3.22)
project(card_compiler LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PIKA_CORE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/pika-core)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${PIKA_CORE}/*.c)
file(GLOB STDLIB_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/vendor/pika-stdlib/*.c)
add_library(card_pika_core STATIC ${CORE_SOURCES} ${STDLIB_SOURCES}
  vendor/pika-api/__pikaBinding.c compiler_context.c)
target_include_directories(card_pika_core PUBLIC ${PIKA_CORE} ${CMAKE_CURRENT_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/vendor/pika-api)
target_compile_definitions(card_pika_core PUBLIC PIKA_CONFIG_ENABLE CROSS_BUILD)
if(MSVC)
  target_compile_definitions(card_pika_core PUBLIC _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
  target_compile_options(card_pika_core PRIVATE /Gy /Gw /utf-8)
else()
  target_compile_options(card_pika_core PRIVATE -ffunction-sections -fdata-sections -fvisibility=hidden)
endif()
if(ANDROID)
  add_library(card_compiler SHARED jni.c)
  target_link_libraries(card_compiler PRIVATE card_pika_core m)
  target_link_options(card_compiler PRIVATE -Wl,--gc-sections)
endif()
add_executable(card-compiler cli.c)
target_link_libraries(card-compiler PRIVATE card_pika_core)
if(MSVC)
  target_link_options(card-compiler PRIVATE /OPT:REF /OPT:ICF)
else()
  target_link_options(card-compiler PRIVATE -Wl,--gc-sections)
  target_link_libraries(card-compiler PRIVATE m)
endif()
